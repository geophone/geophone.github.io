It looks like some modern captchas can be bypassed trivially with opencv.

There's reliance that you won't be able to segment out images and match them.

This is reliably false with Segment Anything 3 by facebook.

I used Segment Anything 2 and found I could extract images and easily match them with a combination of image slicing and predictability of the captcha.

If a captcha asks you in one reliable portion of the image to match images on the left to the right or so forth.  You can use the segmented image, match on the left, and then match on the right with the pixel coordinates easy to find with opencv match.

Here's a link to segment anything 3 (SAM3)[https://github.com/facebookresearch/sam3]

I'll also add some code I used specifically that was mostly generated by claude

```python
"""
Find ALL occurrences of a cutout (PNG with transparent background) inside a
larger scene image — handles the object appearing multiple times, drawing a
SEPARATE box around each individual match.

Approach:
1. Find all good SIFT matches between the cutout and the scene in one pass.
2. Spatially cluster the matched points in the SCENE by proximity (points near
   flamingo #1 form one cluster, points near flamingo #2 form another, etc.).
   This is the key step that keeps instances separate — without it, a single
   homography can accidentally blend points from two different objects and
   produce one box spanning both.
3. Compute a homography independently within each cluster to locate and draw
   that specific instance.

Usage:
    python find_cutout_in_scene.py --cutout flamingo_cutout.png --scene beach_photo.jpg
    python find_cutout_in_scene.py --cutout flamingo_cutout.png --scene beach_photo.jpg --max-instances 5
"""

import argparse
import cv2
import numpy as np


def load_cutout_with_mask(path):
    """Load a RGBA cutout and build a binary mask from its alpha channel."""
    img = cv2.imread(path, cv2.IMREAD_UNCHANGED)
    if img is None:
        raise FileNotFoundError(f"Could not load {path}")

    if img.shape[2] == 4:
        bgr = img[:, :, :3]
        alpha = img[:, :, 3]
        mask = (alpha > 10).astype(np.uint8) * 255  # threshold to avoid faint edge pixels
    else:
        # No alpha channel — treat everything as valid
        bgr = img
        mask = np.full(img.shape[:2], 255, dtype=np.uint8)

    return bgr, mask

from collections import defaultdict
def cluster_points(points, eps):
    """
    Simple distance-based clustering (union-find): any two points closer than
    `eps` get merged into the same cluster. No external dependency needed.
    Returns a list of index-lists, one per cluster.
    """
    n = len(points)
    
    dups = []
    for i in range(n):
        for j in range(i + 1, n):
            if np.linalg.norm(points[i] - points[j]) < eps:
                print(f'noting duplicate {points[j]}')
                dups.append(points[j])
                
    return points[~np.isin(points, dups)].reshape(-1, 2)


def find_all_instances(cutout_path, scene_path, min_matches=8, max_instances=10,
                        ratio_thresh=0.75, ransac_thresh=5.0, cluster_eps=None,
                        output_path="match_result.png"):
    cutout_bgr, cutout_mask = load_cutout_with_mask(cutout_path)
    scene_bgr = cv2.imread(scene_path, cv2.IMREAD_COLOR)
    if scene_bgr is None:
        raise FileNotFoundError(f"Could not load {scene_path}")

    cutout_gray = cv2.cvtColor(cutout_bgr, cv2.COLOR_BGR2GRAY)
    scene_gray = cv2.cvtColor(scene_bgr, cv2.COLOR_BGR2GRAY)
    h, w = cutout_gray.shape

    # SIFT feature detector — restrict cutout keypoints to the masked (opaque) region
    sift = cv2.SIFT_create()
    kp1, des1 = sift.detectAndCompute(cutout_gray, mask=cutout_mask)
    kp2, des2 = sift.detectAndCompute(scene_gray, mask=None)

    if des1 is None or des2 is None or len(kp1) < 2:
        print("Not enough features found in the cutout to match.")
        return []

    print(f"Cutout keypoints: {len(kp1)}   Scene keypoints: {len(kp2)}")

    # Default cluster radius: roughly the size of the cutout itself, since two
    # separate instances of the object should be at least about one object-width
    # apart in the scene. Override with --cluster-eps if your instances are
    # packed closer together than that.
    if cluster_eps is None:
        cluster_eps = 0.75 * np.hypot(w, h)
        print(f'Cluster eps{cluster_eps}')
    cluster_x = 30
    cluster_y = 30

    # --- Step 1: find ALL good matches in one pass (not per-instance) ---
    bf = cv2.BFMatcher(cv2.NORM_L2)
    matches = bf.knnMatch(des1, des2, k=2)

    good = []
    for pair in matches:
        if len(pair) < 2:
            continue
        m, n = pair
        if m.distance < ratio_thresh * n.distance:
            good.append(m)

    print(f"Total good matches across whole scene: {len(good)}")

    if len(good) < min_matches:
        print("Not enough good matches found anywhere in the scene.")
        return []

    dst_points = np.float32([kp2[m.trainIdx].pt for m in good])	
    colors = [(0, 255, 0), (255, 0, 0), (0, 0, 255), (255, 255, 0),
              (255, 0, 255), (0, 255, 255), (128, 255, 0), (0, 128, 255)]
    result = scene_bgr.copy()
    print(dst_points)
    clusters = cluster_points(dst_points, eps=cluster_eps)
    print(clusters)
    for i, cluster in enumerate(clusters):
        print(cluster)
        x, y = cluster
        print(cluster_x)
        print(cluster_y)
        x0 = int(x - cluster_x)
        x1 = int(x + cluster_x)
        y0 = int(y - cluster_y)
        y1 = int(y + cluster_y)
        color = colors[i % len(colors)]
        cv2.rectangle(result, (x0, y0), (x1, y1), color, 3, cv2.LINE_AA)
        label = f"#{i + 1}"
        label_y = y0 - 10 if y0 - 10 > 10 else y0 + 25
        cv2.putText(result, label, (x0, label_y),
                    cv2.FONT_HERSHEY_SIMPLEX, 1.0, color, 2, cv2.LINE_AA)
        #print(f"Instance {i + 1}: bbox (x0,y0,x1,y1) = {f['bbox']}, inliers = {f['inlier_count']}")


    cv2.imwrite(output_path, result)
    return


if __name__ == "__main__":
    parser = argparse.ArgumentParser()
    parser.add_argument("--cutout", required=True, help="Path to transparent-background cutout PNG")
    parser.add_argument("--scene", required=True, help="Path to the larger scene image to search")
    parser.add_argument("--output", default="match_result.png", help="Path to save annotated result")
    parser.add_argument("--min-matches", type=int, default=8,
                         help="Minimum good matches required per cluster to accept an instance")
    parser.add_argument("--max-instances", type=int, default=10,
                         help="Maximum number of instances to search for")
    parser.add_argument("--ratio-thresh", type=float, default=0.75,
                         help="Lowe's ratio test threshold (lower = stricter matches)")
    parser.add_argument("--ransac-thresh", type=float, default=5.0,
                         help="RANSAC reprojection error threshold in pixels")
    parser.add_argument("--cluster-eps", type=float, default=None,
                         help="Max pixel distance between matches to be considered the same "
                              "instance. Default: 0.75x the cutout's diagonal size. Lower this "
                              "if instances are close together and getting merged; raise it if "
                              "a single instance is being split into multiple boxes.")
    args = parser.parse_args()

    find_all_instances(
        args.cutout, args.scene,
        min_matches=args.min_matches,
        max_instances=args.max_instances,
        ratio_thresh=args.ratio_thresh,
        ransac_thresh=args.ransac_thresh,
        cluster_eps=args.cluster_eps,
        output_path=args.output,
    )
```

I heavily modified this code to support finding just two matches, mostly the eps value or the size of the reference image was distorted by transparent background.  When specifying a reasonable height and width of the actual matching image it was easier to cluster matches as duplicates and also tune the threshold so that only two unique matches were found.

There is extensive tuning required of the threshold for a good match given images but I am fairly certian given previous experience there is a sweet spot for matching the extracted images from SAM2

And SAM2 extraction which just worked out of the box, would've been better to use SAM3 but I wasn't aware of it at the time

```python
"""
Extract an object (e.g. a flamingo) from an image using SAM2.

Setup (one time):
    git clone https://github.com/facebookresearch/sam2.git
    cd sam2
    pip install -e .
    cd checkpoints && ./download_ckpts.sh   # downloads sam2 model weights

Usage:
    python sam2_extract_object.py --image flamingo.jpg --point 450 300
    # or with a bounding box instead of a point:
    python sam2_extract_object.py --image flamingo.jpg --box 300 150 650 500
"""

import argparse
import numpy as np
from PIL import Image
import torch

from sam2.build_sam import build_sam2
from sam2.sam2_image_predictor import SAM2ImagePredictor


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--image", required=True, help="Path to input image")
    parser.add_argument("--point", nargs=2, type=int, metavar=("X", "Y"),
                         help="Include point (x y) on the object, e.g. center of the flamingo")
    parser.add_argument("--box", nargs=4, type=int, metavar=("X0", "Y0", "X1", "Y1"),
                         help="Bounding box (x0 y0 x1 y1) around the object")
    parser.add_argument("--checkpoint", default="checkpoints/sam2.1_hiera_large.pt",
                         help="Path to SAM2 checkpoint")
    parser.add_argument("--model-cfg", default="configs/sam2.1/sam2.1_hiera_l.yaml",
                         help="SAM2 model config")
    parser.add_argument("--output", default="flamingo_cutout.png",
                         help="Output PNG path (transparent background)")
    args = parser.parse_args()

    if not args.point and not args.box:
        parser.error("Provide either --point X Y or --box X0 Y0 X1 Y1")

    device = "cuda" if torch.cuda.is_available() else "cpu"
    print(f"Using device: {device}")

    # Load image
    image = Image.open(args.image).convert("RGB")
    image_np = np.array(image)

    # Build SAM2 model + predictor
    sam2_model = build_sam2(args.model_cfg, args.checkpoint, device=device)
    predictor = SAM2ImagePredictor(sam2_model)
    predictor.set_image(image_np)

    # Prepare prompts
    point_coords, point_labels, box = None, None, None
    if args.point:
        point_coords = np.array([args.point])
        point_labels = np.array([1])  # 1 = include/foreground point
    if args.box:
        box = np.array(args.box)

    # Run segmentation
    masks, scores, _ = predictor.predict(
        point_coords=point_coords,
        point_labels=point_labels,
        box=box,
        multimask_output=True,  # returns several candidate masks; we pick the best
    )

    # Pick the highest-confidence mask
    best_idx = np.argmax(scores)
    mask = masks[best_idx].astype(bool)
    print(f"Best mask score: {scores[best_idx]:.3f}")

    # Build an RGBA cutout: keep pixels inside mask, make everything else transparent
    rgba = np.zeros((*image_np.shape[:2], 4), dtype=np.uint8)
    rgba[..., :3] = image_np
    rgba[..., 3] = np.where(mask, 255, 0).astype(np.uint8)

    out_image = Image.fromarray(rgba, mode="RGBA")
    out_image.save(args.output)
    print(f"Saved extracted object to: {args.output}")


if __name__ == "__main__":
    main()
```
